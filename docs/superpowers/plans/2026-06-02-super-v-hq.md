# Super V HQ Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a working Roblox Supervillain HQ tycoon with per-player plots, passive income, drop-pad room progression, and a Robux-powered raid mechanic.

**Architecture:** Server scripts manage plot assignment, income ticks, and raid processing via Roblox services. Client scripts handle proximity prompts and HUD. Shared modules in ReplicatedStorage hold config and RemoteEvents so server and client stay decoupled. Rojo syncs all `.luau` files from disk into Studio live.

**Tech Stack:** Luau, Rojo 7.x (Linux binary for WSL2), Roblox Studio (Windows), DataStoreService, MarketplaceService, GitHub (`les250/super-v-hq`)

**Testing note:** Roblox scripts depend on game services that only run inside Studio. Pure data modules (Config, RoomData) can be verified with the standalone `luau` binary. All other scripts include an in-Studio acceptance test at the end of their task.

**Item trading** (mentioned in spec economy table) is deferred to a follow-up plan — the core loop is complete without it.

        unlockOrder  = 6,

            { cost = 20000,  incomePerTick = 80  },
            { cost = 60000,  incomePerTick = 180 },
            { cost = 200000, incomePerTick = 400 },
        },
    },
}

-- Returns the room table for a given id, or nil
function RoomData.getRoom(id: string)
    for _, room in ipairs(RoomData.ROOMS) do
        if room.id == id then
            return room
        end
    end
    return nil
end

-- Returns the vault capacity for a player given their built rooms
-- builtRooms: { [roomId: string]: tier: number } where tier is 1-3
function RoomData.getVaultCap(builtRooms: {[string]: number}, baseCap: number): number
    local cap = baseCap
    local vault = RoomData.getRoom("vault")
    if vault then
        local tier = builtRooms["vault"]
        if tier and tier > 0 then
            cap += vault.vaultCapBonus[tier]
        end
    end
    return cap
end

-- Returns the total defense reduction fraction for a player's built rooms
-- e.g. 0.07 means 7% is subtracted from RAID_STEAL_PERCENT
function RoomData.getDefenseReduction(builtRooms: {[string]: number}): number
    local reduction = 0
    for _, room in ipairs(RoomData.ROOMS) do
        if room.raidDefenseBonus then
            local tier = builtRooms[room.id]
            if tier and tier > 0 then
                reduction += room.raidDefenseBonus[tier]
            end
        end
    end
    return reduction
end

-- Returns total incomePerTick across all built rooms
function RoomData.getTotalIncome(builtRooms: {[string]: number}): number
    local total = 0
    for _, room in ipairs(RoomData.ROOMS) do
        local tier = builtRooms[room.id]
        if tier and tier > 0 then
            total += room.tiers[tier].incomePerTick
        end
    end
    return total
end

return RoomData
```

local ReplicatedStorage = game:GetService("ReplicatedStorage")

local function getOrCreate(className: string, name: string)
    local existing = ReplicatedStorage:FindFirstChild(name)
    if existing then
        return existing
    end
    local obj = Instance.new(className)
    obj.Name = name
    obj.Parent = ReplicatedStorage
    return obj

local Remotes = {}

  lastRaidedAt: number,                 -- os.time() of last raid received
}
```

- [ ] **Step 1: Write DataStore.server.luau**

```lua
-- src/ServerScriptService/DataStore.server.luau
local DataStoreService = game:GetService("DataStoreService")
local Players          = game:GetService("Players")
local Config           = require(game.ReplicatedStorage.Config)

local store = DataStoreService:GetDataStore("SuperVHQ_v1")

-- In-memory cache: { [userId: number]: playerData }
local cache: {[number]: any} = {}

local DEFAULT_DATA = {
    currency     = Config.STARTING_CURRENCY,
    rooms        = {},
    vaultAmount  = 0,
    lastRaidedAt = 0,
}

local function deepCopy(t)
    local copy = {}
    for k, v in pairs(t) do
        copy[k] = type(v) == "table" and deepCopy(v) or v
    end
    return copy
end

local function loadData(player: Player)
    local key = "Player_" .. player.UserId
    local success, data = pcall(function()
        return store:GetAsync(key)
    end)
    if success and data then
        -- Merge with defaults so new fields appear for existing players
        local merged = deepCopy(DEFAULT_DATA)
        for k, v in pairs(data) do
            merged[k] = v
        end
        cache[player.UserId] = merged
    else
        cache[player.UserId] = deepCopy(DEFAULT_DATA)
    end
end

local function saveData(player: Player)
    local data = cache[player.UserId]
    if not data then return end
    local key = "Player_" .. player.UserId
    local success, err = pcall(function()
        store:SetAsync(key, data)
    end)
    if not success then
        warn("DataStore save failed for", player.Name, err)
    end
end

-- Public API used by other server scripts
local DataStore = {}

function DataStore.get(player: Player)
    return cache[player.UserId]
end

function DataStore.save(player: Player)
    saveData(player)
end

-- Auto-load on join, auto-save on leave and every 60 seconds
Players.PlayerAdded:Connect(function(player)
    loadData(player)
end)

Players.PlayerRemoving:Connect(function(player)
    saveData(player)
    cache[player.UserId] = nil
end)

-- Periodic auto-save
task.spawn(function()
    while true do
        task.wait(60)
        for _, player in ipairs(Players:GetPlayers()) do
            saveData(player)
        end
    end
end)

-- Load data for anyone already in the game (Studio playtest scenario)
for _, player in ipairs(Players:GetPlayers()) do
    loadData(player)
end

_G.DataStore = DataStore
```

## Task 6: PlotManager server script

**Files:**
- Create: `src/ServerScriptService/PlotManager.server.luau`

Plots are Folder instances named `Plot_1` through `Plot_10` in `Workspace`. Each has a `BoolValue` named `IsOccupied` and an `IntValue` named `OwnerId`. These are created in Studio and saved in the `.rbxlx` file (not synced by Rojo — see Task 11 for Studio setup).

- [ ] **Step 1: Write PlotManager.server.luau**

```lua
-- src/ServerScriptService/PlotManager.server.luau
local Players   = game:GetService("Players")
local Workspace = game:GetService("Workspace")

-- { [userId: number]: plotFolder }
local playerPlots: {[number]: Folder} = {}

local function getPlots(): {Folder}
    local plots = {}
    for _, obj in ipairs(Workspace:GetChildren()) do
        if obj:IsA("Folder") and obj.Name:match("^Plot_%d+$") then
            table.insert(plots, obj)
        end
    end
    table.sort(plots, function(a, b) return a.Name < b.Name end)
    return plots
end

local function assignPlot(player: Player): Folder?
    for _, plot in ipairs(getPlots()) do
        local occupied = plot:FindFirstChild("IsOccupied")
        if occupied and not occupied.Value then
            occupied.Value = true
            local ownerId = plot:FindFirstChild("OwnerId")
            if ownerId then
                ownerId.Value = player.UserId
            end
            playerPlots[player.UserId] = plot
            return plot
        end
    end
    return nil
end

local function freePlot(player: Player)
    local plot = playerPlots[player.UserId]
    if plot then
        local occupied = plot:FindFirstChild("IsOccupied")
        if occupied then occupied.Value = false end
        local ownerId = plot:FindFirstChild("OwnerId")
        if ownerId then ownerId.Value = 0 end
        playerPlots[player.UserId] = nil
    end
end

-- Public API
local PlotManager = {}

function PlotManager.getPlot(player: Player): Folder?
    return playerPlots[player.UserId]
end

function PlotManager.getPlotByUserId(userId: number): Folder?
    return playerPlots[userId]
end

Players.PlayerAdded:Connect(function(player)
    local plot = assignPlot(player)
    if not plot then
        warn("No free plot for", player.Name)
    end
end)

Players.PlayerRemoving:Connect(function(player)
    freePlot(player)
end)

for _, player in ipairs(Players:GetPlayers()) do
    assignPlot(player)
end

_G.PlotManager = PlotManager
```

```lua
-- src/ServerScriptService/EconomyManager.server.luau
local Players   = game:GetService("Players")
local Config    = require(game.ReplicatedStorage.Config)
local RoomData  = require(game.ReplicatedStorage.RoomData)
local Remotes   = require(game.ReplicatedStorage.Remotes)

local function getDS(): any
    -- Wait until DataStore is ready (loaded by DataStore.server.luau)
    while not _G.DataStore do task.wait(0.1) end
    return _G.DataStore
end

local function pushCurrencyUpdate(player: Player)
    local ds   = getDS()
    local data = ds.get(player)
    if not data then return end
    local cap = RoomData.getVaultCap(data.rooms, Config.BASE_VAULT_CAP)
    Remotes.UpdateCurrency:FireClient(player, data.vaultAmount, cap)
end

-- Passive income tick
task.spawn(function()
    while true do
        task.wait(Config.INCOME_TICK_INTERVAL)
        local ds = getDS()
        for _, player in ipairs(Players:GetPlayers()) do
            local data = ds.get(player)
            if data then
                local income = RoomData.getTotalIncome(data.rooms)
                local cap    = RoomData.getVaultCap(data.rooms, Config.BASE_VAULT_CAP)
                data.vaultAmount = math.min(data.vaultAmount + income, cap)
                pushCurrencyUpdate(player)
            end
        end
    end
end)

-- Buy a room (drop pad activated)
Remotes.BuyRoom.OnServerEvent:Connect(function(player: Player, roomId: string)
    local ds   = getDS()
    local data = ds.get(player)
    if not data then return end

    local room = RoomData.getRoom(roomId)
    if not room then return end

    -- Must not already own this room
    if data.rooms[roomId] and data.rooms[roomId] > 0 then return end

    -- Must own all previous rooms first
    for _, r in ipairs(RoomData.ROOMS) do
        if r.unlockOrder < room.unlockOrder then
            if not data.rooms[r.id] or data.rooms[r.id] == 0 then
                return -- prerequisite not met
            end
        end
    end

    local cost = room.tiers[1].cost
    if data.vaultAmount < cost then return end

    data.vaultAmount   -= cost
    data.rooms[roomId]  = 1
    pushCurrencyUpdate(player)
    Remotes.RoomBuilt:FireClient(player, roomId, 1)
end)

-- Upgrade a room (room upgrade menu activated)
Remotes.UpgradeRoom.OnServerEvent:Connect(function(player: Player, roomId: string)
    local ds   = getDS()
    local data = ds.get(player)
    if not data then return end

    local room        = RoomData.getRoom(roomId)
    local currentTier = data.rooms[roomId] or 0
    if not room or currentTier == 0 or currentTier >= 3 then return end

    local nextTier = currentTier + 1
    local cost     = room.tiers[nextTier].cost
    if data.vaultAmount < cost then return end

    data.vaultAmount  -= cost
    data.rooms[roomId] = nextTier
    pushCurrencyUpdate(player)
    Remotes.RoomBuilt:FireClient(player, roomId, nextTier)
end)

-- GetPlayerData remote function
Remotes.GetPlayerData.OnServerInvoke = function(player: Player)
    local ds = getDS()
    return ds.get(player)
end

-- GetAllPlotInfo (used by War Room ability)
Remotes.GetAllPlotInfo.OnServerInvoke = function(player: Player)
    local ds       = getDS()
    local data     = ds.get(player)
    -- Only players with War Room built can call this
    if not data or not data.rooms["war_room"] or data.rooms["war_room"] == 0 then
        return {}
    end
    local result = {}
    for _, p in ipairs(Players:GetPlayers()) do
        local pData = ds.get(p)
        if pData then
            table.insert(result, {
                userId      = p.UserId,
                name        = p.Name,
                vaultAmount = pData.vaultAmount,
            })
        end
    end
    return result
end
```

1. Client fires `PrepareRaid` with the target's userId
2. Server stores `pendingRaids[raiderUserId] = targetUserId`
3. Client calls `MarketplaceService:PromptProductPurchase(player, Config.PRODUCT_RAID)`
4. Roblox calls `ProcessReceipt` on the server
5. Server reads `pendingRaids` to find the target and executes the steal

- [ ] **Step 1: Write RaidManager.server.luau**

```lua
-- src/ServerScriptService/RaidManager.server.luau
local Players            = game:GetService("Players")
local MarketplaceService = game:GetService("MarketplaceService")
local Config             = require(game.ReplicatedStorage.Config)
local RoomData           = require(game.ReplicatedStorage.RoomData)
local Remotes            = require(game.ReplicatedStorage.Remotes)

local function getDS(): any
    while not _G.DataStore do task.wait(0.1) end
    return _G.DataStore
end

-- Stores which player each raider is targeting
-- { [raiderUserId: number]: targetUserId: number }
local pendingRaids: {[number]: number} = {}

-- Client signals intent to raid before prompting purchase
Remotes.PrepareRaid.OnServerEvent:Connect(function(raider: Player, targetUserId: number)
    -- Validate target exists in server
    local targetPlayer = Players:GetPlayerByUserId(targetUserId)
    if not targetPlayer or targetPlayer == raider then return end
    pendingRaids[raider.UserId] = targetUserId
end)

local function executeRaid(raider: Player, targetUserId: number)
    local ds         = getDS()
    local targetPlayer = Players:GetPlayerByUserId(targetUserId)
    if not targetPlayer then return end

    local targetData = ds.get(targetPlayer)
    if not targetData then return end

    -- Check cooldown
    local now = os.time()
    if (now - targetData.lastRaidedAt) < Config.RAID_COOLDOWN then
        -- Target is still in cooldown — refund not possible via Roblox API,
        -- but we silently abort. Client should check cooldown before prompting.
        return
    end

    local defense      = RoomData.getDefenseReduction(targetData.rooms)
    local stealPercent = math.max(0, Config.RAID_STEAL_PERCENT - defense)
    local stolen       = math.floor(targetData.vaultAmount * stealPercent)

    targetData.vaultAmount  = math.max(0, targetData.vaultAmount - stolen)
    targetData.lastRaidedAt = now

    -- Give stolen amount to raider
    local raiderData = ds.get(raider)
    if raiderData then
        local cap = RoomData.getVaultCap(raiderData.rooms, Config.BASE_VAULT_CAP)
        raiderData.vaultAmount = math.min(raiderData.vaultAmount + stolen, cap)
        Remotes.UpdateCurrency:FireClient(raider, raiderData.vaultAmount, cap)
    end

    -- Notify victim
    local targetCap = RoomData.getVaultCap(targetData.rooms, Config.BASE_VAULT_CAP)
    Remotes.UpdateCurrency:FireClient(targetPlayer, targetData.vaultAmount, targetCap)
    Remotes.RaidNotification:FireClient(targetPlayer, raider.Name, stolen)
end

-- Handle currency product purchases and raid purchases
MarketplaceService.ProcessReceipt = function(receiptInfo)
    local ds     = getDS()
    local player = Players:GetPlayerByUserId(receiptInfo.PlayerId)
    if not player then
        return Enum.ProductPurchaseDecision.NotProcessedYet
    end

    local data = ds.get(player)
    if not data then
        return Enum.ProductPurchaseDecision.NotProcessedYet
    end

    local productId = receiptInfo.ProductId

    -- Currency purchases
    local currencyAmount = Config.PRODUCT_CURRENCY_AMOUNTS[productId]
    if currencyAmount then
        local cap = RoomData.getVaultCap(data.rooms, Config.BASE_VAULT_CAP)
        data.vaultAmount = math.min(data.vaultAmount + currencyAmount, cap)
        Remotes.UpdateCurrency:FireClient(player, data.vaultAmount, cap)
        ds.save(player)
        return Enum.ProductPurchaseDecision.PurchaseGranted
    end

    -- Raid purchase
    if productId == Config.PRODUCT_RAID then
        local targetUserId = pendingRaids[player.UserId]
        if targetUserId then
            pendingRaids[player.UserId] = nil
            executeRaid(player, targetUserId)
        end
        ds.save(player)
        return Enum.ProductPurchaseDecision.PurchaseGranted
    end

    return Enum.ProductPurchaseDecision.NotProcessedYet
end
```

## Task 9: DropPadClient LocalScript

**Files:**
- Create: `src/StarterPlayerScripts/DropPadClient.client.luau`

Drop pads are `Part` instances inside the player's plot folder, named `DropPad_lab`, `DropPad_armory`, etc. (one per room). They have a `ProximityPrompt` child added by this script. After a room is bought, the drop pad for that room is hidden and a `DropPad_upgrade_<roomId>` prompt appears inside the built room model.

- [ ] **Step 1: Write DropPadClient.client.luau**

```lua
-- src/StarterPlayerScripts/DropPadClient.client.luau
local Players          = game:GetService("Players")
local Remotes          = require(game.ReplicatedStorage.Remotes)

local player    = Players.LocalPlayer
local character = player.Character or player.CharacterAdded:Wait()

-- Wait for PlotManager to assign us a plot
local function getMyPlot(): Folder
    while true do
        for _, obj in ipairs(workspace:GetChildren()) do
            if obj:IsA("Folder") and obj.Name:match("^Plot_%d+$") then
                local ownerId = obj:FindFirstChild("OwnerId")
                if ownerId and ownerId.Value == player.UserId then
                    return obj
                end
            end
        end
        task.wait(0.5)
    end
end

local plot = getMyPlot()

-- Add ProximityPrompt to a drop pad part
local function setupDropPad(part: Part, roomId: string)
    local prompt = Instance.new("ProximityPrompt")
    prompt.ActionText    = "Build " .. roomId:gsub("_", " "):gsub("^%l", string.upper)
    prompt.ObjectText    = "Drop Pad"
    prompt.HoldDuration  = 0.5
    prompt.MaxActivationDistance = 8
    prompt.Parent = part

    prompt.Triggered:Connect(function()
        Remotes.BuyRoom:FireServer(roomId)
    end)
end

-- Add upgrade prompt inside a built room model
local function setupUpgradePad(roomModel: Model, roomId: string)
    local upgradePart = Instance.new("Part")
    upgradePart.Name      = "UpgradePad"
    upgradePart.Size      = Vector3.new(4, 0.5, 4)
    upgradePart.Anchored  = true
    upgradePart.BrickColor = BrickColor.new("Bright yellow")
    upgradePart.Parent    = roomModel

    -- Position at room centre, on the floor
    upgradePart.CFrame = roomModel:GetPivot() * CFrame.new(0, -roomModel:GetExtentsSize().Y / 2, 0)

    local prompt = Instance.new("ProximityPrompt")
    prompt.ActionText   = "Upgrade"
    prompt.ObjectText   = roomId:gsub("_", " "):gsub("^%l", string.upper)
    prompt.HoldDuration = 0.5
    prompt.MaxActivationDistance = 8
    prompt.Parent = upgradePart

    prompt.Triggered:Connect(function()
        Remotes.UpgradeRoom:FireServer(roomId)
    end)
end

-- Wire up all drop pads found in this plot
for _, obj in ipairs(plot:GetDescendants()) do
    if obj:IsA("Part") then
        local roomId = obj.Name:match("^DropPad_(.+)$")
        if roomId then
            setupDropPad(obj, roomId)
        end
    end
end

-- When a room is built, hide its drop pad, show the pre-placed room model, add upgrade pad
-- Room models are pre-placed in Studio as Model "Room_<roomId>" with all parts set
-- Transparency=1 so they are invisible until bought.
Remotes.RoomBuilt.OnClientEvent:Connect(function(roomId: string, tier: number)
    -- Hide the drop pad for this room
    for _, obj in ipairs(plot:GetDescendants()) do
        if obj.Name == "DropPad_" .. roomId then
            obj.Transparency = 1
            local prompt = obj:FindFirstChildOfClass("ProximityPrompt")
            if prompt then prompt.Enabled = false end
        end
    end

    -- Make the pre-placed room model visible (all parts → Transparency 0)
    local roomModel = plot:FindFirstChild("Room_" .. roomId)
    if roomModel and roomModel:IsA("Model") then
        for _, part in ipairs(roomModel:GetDescendants()) do
            if part:IsA("BasePart") then
                part.Transparency = 0
            end
        end
        -- If tier 1 just built, add the upgrade proximity prompt
        if tier == 1 then
            setupUpgradePad(roomModel, roomId)
        end
    end
end)
```

toastLabel.TextColor3       = Color3.fromRGB(255, 255, 255)
toastLabel.TextScaled       = true
toastLabel.Font             = Enum.Font.Gotham

toastLabel.Parent           = toastFrame

local function showToast(message: string)
    toastLabel.Text = message
    -- Slide in
    TweenService:Create(toastFrame,
        TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
        { Position = UDim2.new(0.5, -200, 0, 16) }
    ):Play()
    -- Slide out after 3.5 seconds
    task.delay(3.5, function()
        TweenService:Create(toastFrame,
            TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.In),
            { Position = UDim2.new(0.5, -200, 0, -70) }
        ):Play()
    end)
end

Remotes.UpdateCurrency.OnClientEvent:Connect(function(amount: number, cap: number)
    currencyLabel.Text = string.format("💰 %d / %d", math.floor(amount), cap)
end)

Remotes.RaidNotification.OnClientEvent:Connect(function(raiderName: string, stolen: number)
    showToast(string.format("⚠️ %s raided your HQ! -%d currency", raiderName, stolen))
end)

task.spawn(function()
    local data = Remotes.GetPlayerData:InvokeServer()
    if data then
        -- Trigger a currency update by re-firing from server is cleaner,
        -- but as fallback read local data
        local Config   = require(game.ReplicatedStorage.Config)
        local RoomData = require(game.ReplicatedStorage.RoomData)
        local cap      = RoomData.getVaultCap(data.rooms, Config.BASE_VAULT_CAP)
        currencyLabel.Text = string.format("💰 %d / %d", math.floor(data.vaultAmount), cap)
    end
end)
